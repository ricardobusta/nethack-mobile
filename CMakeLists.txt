cmake_minimum_required(VERSION 3.22)
project(nethack_mobile LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(NH_SOURCE "${CMAKE_SOURCE_DIR}/build/upstream" CACHE PATH "Patched generated NetHack source")
option(NH_TOOLS_ONLY "Build host data generators" OFF)
option(NH_GODOT "Build the Godot extension" ON)
set(NH_FLAGS NOTTYGRAPHICS SHIM_GRAPHICS LIBNH NOSHELL NOMAIL NO_SIGNAL NOPANICTRACE NOCRASHREPORT NO_CHRONICLE NOCWD_ASSUMPTIONS DLB ZLIB_COMP TILES_IN_GLYPHMAP)
include_directories("${NH_SOURCE}/include" "${CMAKE_SOURCE_DIR}/vendor/lua/src")
if(NH_TOOLS_ONLY)
  add_compile_definitions(${NH_FLAGS})
  add_executable(makedefs ${NH_SOURCE}/util/makedefs.c ${NH_SOURCE}/util/panic.c
    ${NH_SOURCE}/src/alloc.c ${NH_SOURCE}/src/monst.c ${NH_SOURCE}/src/objects.c
    ${NH_SOURCE}/src/date.c ${NH_SOURCE}/src/hacklib.c)
  add_executable(tilemap ${NH_SOURCE}/win/share/tilemap.c ${NH_SOURCE}/src/objects.c
    ${NH_SOURCE}/src/monst.c ${NH_SOURCE}/src/drawing.c ${NH_SOURCE}/src/hacklib.c ${NH_SOURCE}/util/panic.c)
  return()
endif()
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
file(GLOB LUA_FILES CONFIGURE_DEPENDS vendor/lua/src/*.c)
list(FILTER LUA_FILES EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(nh_lua STATIC ${LUA_FILES})
target_link_libraries(nh_lua PUBLIC m ${CMAKE_DL_LIBS})
# Read upstream's authoritative core object list instead of guessing source files.
file(READ "${NH_SOURCE}/sys/unix/Makefile.src" NH_MAKEFILE)
string(REGEX MATCH "HOBJ = [^#]+#" NH_OBJECTS "${NH_MAKEFILE}")
string(REGEX MATCHALL "[a-z0-9_]+\\.o" NH_OBJECTS "${NH_OBJECTS}")
set(NH_FILES)
foreach(OBJ IN LISTS NH_OBJECTS)
  string(REPLACE ".o" ".c" SRC "${OBJ}")
  if(EXISTS "${NH_SOURCE}/src/${SRC}")
    list(APPEND NH_FILES "${NH_SOURCE}/src/${SRC}")
  endif()
endforeach()
list(APPEND NH_FILES ${NH_SOURCE}/src/hacklib.c ${NH_SOURCE}/src/date.c ${NH_SOURCE}/src/tile.c
  ${NH_SOURCE}/sys/unix/unixmain.c ${NH_SOURCE}/sys/unix/unixunix.c ${NH_SOURCE}/sys/unix/unixres.c
  ${NH_SOURCE}/sys/share/posixregex.c ${NH_SOURCE}/sys/share/random.c ${NH_SOURCE}/win/shim/winshim.c)
list(REMOVE_DUPLICATES NH_FILES)
set_source_files_properties(${NH_SOURCE}/sys/unix/unixmain.c PROPERTIES COMPILE_DEFINITIONS "main=nhmain")
add_library(nh_engine STATIC ${NH_FILES})
target_compile_definitions(nh_engine PRIVATE ${NH_FLAGS} NH_MOBILE exit=nm_engine_exit DEFAULT_WINDOW_SYS="shim" HACKDIR=".")
target_compile_options(nh_engine PRIVATE -Wno-deprecated-non-prototype -Wno-format)
target_link_libraries(nh_engine PUBLIC nh_lua ZLIB::ZLIB Threads::Threads)
add_library(nh_mobile STATIC native/mobile.c)
target_compile_definitions(nh_mobile PRIVATE ${NH_FLAGS} NH_MOBILE DEFAULT_WINDOW_SYS="shim" HACKDIR=".")
target_include_directories(nh_mobile PUBLIC native)
target_link_libraries(nh_mobile PUBLIC nh_engine)
if(NOT ANDROID)
  add_executable(nh_smoke native/smoke.c)
  target_link_libraries(nh_smoke PRIVATE nh_mobile)
endif()
if(NH_GODOT)
  set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_SOURCE_DIR}/native/godot_profile.json" CACHE PATH "" FORCE)
  add_subdirectory(vendor/godot-cpp EXCLUDE_FROM_ALL)
  add_library(nethack_mobile SHARED native/extension.cpp)
  target_link_libraries(nethack_mobile PRIVATE nh_mobile godot-cpp)
  if(ANDROID)
    target_link_options(nethack_mobile PRIVATE "-Wl,-z,max-page-size=16384")
  endif()
  set_target_properties(nethack_mobile PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/godot/bin/${CMAKE_SYSTEM_NAME}/${CMAKE_SYSTEM_PROCESSOR}")
endif()
